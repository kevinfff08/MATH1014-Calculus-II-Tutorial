\section{Review}

\subsection{Power Series and Interval of Convergence}
A \textbf{power series} about $x = a$ is a series of the form:
\[ \sum_{n=0}^{\infty} c_n (x-a)^n = c_0 + c_1(x-a) + c_2(x-a)^2 + \dots \]
\begin{itemize}
    \item \textbf{Radius of Convergence ($R$):} We typically use the Ratio Test or Root Test to find $R$.
    By solving $\lim_{n \to \infty} \left| \frac{c_{n+1}(x-a)^{n+1}}{c_n(x-a)^n} \right| < 1$, we get $|x-a| < R$.
    \item \textbf{Interval of Convergence:} The series converges absolutely for $x \in (a-R, a+R)$.
    \item \textbf{Endpoints:} The Ratio/Root tests are inconclusive at the boundaries $x = a-R$ and $x = a+R$. You \textbf{must} manually plug these values into the original series and use other tests (e.g., Alternating Series Test, $p$-series) to determine if they are included in the interval.
\end{itemize}

\subsection{Taylor and Maclaurin Series}
If $f(x)$ has derivatives of all orders at $x=a$, the \textbf{Taylor Series} of $f(x)$ centered at $x=a$ is:
\[ f(x) = \sum_{n=0}^{\infty} \frac{f^{(n)}(a)}{n!} (x-a)^n = f(a) + f'(a)(x-a) + \frac{f''(a)}{2!}(x-a)^2 + \dots \]
When $a=0$, it is called the \textbf{Maclaurin Series}. You should memorize these common Maclaurin Series:
\begin{itemize}
    \item $\displaystyle \frac{1}{1-x} = \sum_{n=0}^{\infty} x^n = 1 + x + x^2 + \dots \quad \text{for } |x| < 1$
    \item $\displaystyle e^x = \sum_{n=0}^{\infty} \frac{x^n}{n!} = 1 + x + \frac{x^2}{2!} + \dots \quad \text{for all } x$
    \item $\displaystyle \ln(1+x) = \sum_{n=1}^{\infty} (-1)^{n-1} \frac{x^n}{n} = x - \frac{x^2}{2} + \frac{x^3}{3} - \dots \quad \text{for } x \in (-1, 1]$
\end{itemize}

\subsection{Operations on Power Series}
Within its interval of convergence, a power series can be treated like a polynomial.
\begin{itemize}
    \item \textbf{Substitution:} You can substitute $x$ with $2x$, $-x^2$, etc., into a known series to find a new series.
    \item \textbf{Differentiation and Integration:} You can differentiate or integrate a power series term-by-term.
    \[ \frac{d}{dx} \left[ \sum c_n(x-a)^n \right] = \sum n c_n (x-a)^{n-1} \]
    \[ \int \left[ \sum c_n(x-a)^n \right] dx = C + \sum c_n \frac{(x-a)^{n+1}}{n+1} \]
    \textbf{Important:} The radius of convergence $R$ remains the same after differentiation or integration, but convergence at the endpoints might change!
\end{itemize}

\section{Exercises}

\subsection{Finding Taylor Expansions and Intervals of Convergence}

\begin{exercise}
Find the Taylor expansion of the following functions at the given point and determine the interval of convergence.
\begin{enumerate}
    \item $\displaystyle \frac{x}{2-x-x^2}$ at $x=0$
    \item $\displaystyle \frac{x-1}{x+1}$ at $x=1$
    \item $\displaystyle (1+x)\ln(1-x)$ at $x=0$
\end{enumerate}
\end{exercise}

\begin{solution}
\textbf{(1)} \textbf{Idea.} Factor the denominator, use partial fractions, and apply the geometric series expansion $\frac{1}{1-r} = \sum r^n$.

First, factor the denominator: $2-x-x^2 = (2+x)(1-x)$.
Now, set up the partial fraction decomposition:
\[
\frac{x}{(2+x)(1-x)} = \frac{A}{1-x} + \frac{B}{2+x} \implies x = A(2+x) + B(1-x)
\]
Setting $x=1 \implies 1 = 3A \implies A = 1/3$. \\
Setting $x=-2 \implies -2 = 3B \implies B = -2/3$.
Thus, the function can be rewritten as:
\[
f(x) = \frac{1}{3}\left( \frac{1}{1-x} \right) - \frac{2}{3}\left( \frac{1}{2+x} \right)
\]
We expand each term into a Maclaurin series.
For the first term:
\[
\frac{1}{1-x} = \sum_{n=0}^{\infty} x^n \quad \text{for } |x| < 1
\]
For the second term, we must rewrite it to look like $\frac{1}{1-r}$:
\[
\frac{1}{2+x} = \frac{1}{2(1 + \frac{x}{2})} = \frac{1}{2} \cdot \frac{1}{1 - (-\frac{x}{2})} = \frac{1}{2} \sum_{n=0}^{\infty} \left(-\frac{x}{2}\right)^n = \sum_{n=0}^{\infty} \frac{(-1)^n}{2^{n+1}} x^n \quad \text{for } \left|-\frac{x}{2}\right| < 1 \implies |x| < 2
\]
Combining both series:
\begin{align*}
f(x) &= \frac{1}{3} \sum_{n=0}^{\infty} x^n - \frac{2}{3} \sum_{n=0}^{\infty} \frac{(-1)^n}{2^{n+1}} x^n \\
&= \frac{1}{3} \sum_{n=0}^{\infty} \left[ 1 - 2 \cdot \frac{(-1)^n}{2^{n+1}} \right] x^n \\
&= \frac{1}{3} \sum_{n=0}^{\infty} \left[ 1 + \frac{(-1)^{n+1}}{2^n} \right] x^n
\end{align*}
\textbf{Interval of Convergence:} 
The first series converges for $|x| < 1$ and the second for $|x| < 2$. The combined series converges on the intersection, which is $|x| < 1$, so $R = 1$.
Now we check the endpoints $x=1$ and $x=-1$:
\begin{itemize}
    \item If $x=1$: The series becomes $\frac{1}{3} \sum_{n=0}^{\infty} \left[ 1 + \frac{(-1)^{n+1}}{2^n} \right] (1)^n$. As $n \to \infty$, the $n$-th term approaches $1 \neq 0$. By the Test for Divergence, it diverges.
    \item If $x=-1$: The series becomes $\frac{1}{3} \sum_{n=0}^{\infty} \left[ 1 + \frac{(-1)^{n+1}}{2^n} \right] (-1)^n$. As $n \to \infty$, the term does not approach 0 (it oscillates near $\pm 1$). It diverges.
\end{itemize}
Thus, the interval of convergence is $(-1, 1)$.

\textbf{(2)} \textbf{Idea.} We need the expansion at $x=1$, which means powers of $(x-1)$. We algebraically manipulate the function to isolate $(x-1)$.

\[
f(x) = \frac{x-1}{x+1} = \frac{x-1}{(x-1) + 2} = \frac{x-1}{2} \cdot \frac{1}{1 + \frac{x-1}{2}}
\]
Now we use the geometric series formula $\frac{1}{1-r} = \sum r^n$ with $r = -\frac{x-1}{2}$:
\begin{align*}
f(x) &= \frac{x-1}{2} \sum_{n=0}^{\infty} \left( -\frac{x-1}{2} \right)^n \\
&= \frac{x-1}{2} \sum_{n=0}^{\infty} \frac{(-1)^n (x-1)^n}{2^n} \\
&= \sum_{n=0}^{\infty} \frac{(-1)^n (x-1)^{n+1}}{2^{n+1}}
\end{align*}
To make the sum look like standard powers of $n$, let $k = n+1$. When $n=0$, $k=1$:
\[
f(x) = \sum_{k=1}^{\infty} \frac{(-1)^{k-1} (x-1)^k}{2^k}
\]
\textbf{Interval of Convergence:}
The geometric series converges when $|r| < 1 \implies \left| -\frac{x-1}{2} \right| < 1 \implies |x-1| < 2$. Thus $R=2$. The open interval is $(-1, 3)$.
Now check the endpoints:
\begin{itemize}
    \item If $x=3$: Series is $\sum_{k=1}^{\infty} \frac{(-1)^{k-1} (2)^k}{2^k} = \sum_{k=1}^{\infty} (-1)^{k-1}$. The limit of the terms is not 0, so it diverges.
    \item If $x=-1$: Series is $\sum_{k=1}^{\infty} \frac{(-1)^{k-1} (-2)^k}{2^k} = \sum_{k=1}^{\infty} (-1)^{k-1} (-1)^k = \sum_{k=1}^{\infty} (-1)^{2k-1} = \sum_{k=1}^{\infty} -1$. It diverges.
\end{itemize}
Thus, the interval of convergence is $(-1, 3)$.

\textbf{(3)} \textbf{Idea.} Use the known Maclaurin series for $\ln(1-x)$ and multiply it by $(1+x)$.

Recall the standard series:
\[
\ln(1+x) = \sum_{n=1}^{\infty} \frac{(-1)^{n-1} x^n}{n}
\]
Substituting $x$ with $-x$:
\[
\ln(1-x) = \sum_{n=1}^{\infty} \frac{(-1)^{n-1} (-x)^n}{n} = \sum_{n=1}^{\infty} \frac{(-1)^{n-1} (-1)^n x^n}{n} = \sum_{n=1}^{\infty} -\frac{x^n}{n}
\]
Now multiply by $(1+x)$:
\begin{align*}
(1+x)\ln(1-x) &= (1+x) \sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^n \\
&= \sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^n + x \sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^n \\
&= \sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^n + \sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^{n+1}
\end{align*}
To combine them, shift the index of the second sum. Let $k = n+1 \implies n = k-1$. When $n=1, k=2$:
\[
\sum_{n=1}^{\infty} \left(-\frac{1}{n}\right)x^{n+1} = \sum_{k=2}^{\infty} \left(-\frac{1}{k-1}\right)x^k
\]
Now pull out the $n=1$ term from the first sum so both sums start at 2:
\begin{align*}
(1+x)\ln(1-x) &= -x + \sum_{n=2}^{\infty} \left(-\frac{1}{n}\right)x^n + \sum_{n=2}^{\infty} \left(-\frac{1}{n-1}\right)x^n \\
&= -x - \sum_{n=2}^{\infty} \left( \frac{1}{n} + \frac{1}{n-1} \right) x^n
\end{align*}
\textbf{Interval of Convergence:}
The standard series for $\ln(1-x)$ has $R=1$, so the open interval is $(-1, 1)$.
Check the endpoints:
\begin{itemize}
    \item If $x=1$: The series evaluates to $-1 - \sum_{n=2}^{\infty} \left( \frac{1}{n} + \frac{1}{n-1} \right)$. Since $\sum \frac{1}{n}$ is a divergent harmonic series, this sum diverges. (Also, $\ln(1-x)$ is undefined at $x=1$).
    \item If $x=-1$: The series evaluates to $1 - \sum_{n=2}^{\infty} \left( \frac{1}{n} + \frac{1}{n-1} \right)(-1)^n$. This is an alternating series. The terms $b_n = \frac{1}{n} + \frac{1}{n-1}$ are decreasing and limit to 0. By the AST, it converges.
\end{itemize}
Thus, the interval of convergence is $[-1, 1)$.
\end{solution}

\subsection{Evaluating Series Sums using Taylor Series Operations}

\begin{exercise}
Use the Taylor expansion of $\frac{e^x-1}{x}$ at $x=0$ to show $\displaystyle \sum_{n=1}^{\infty}\frac{n}{(n+1)!}=1$.
\end{exercise}

\begin{solution}
\textbf{Idea.} Start with the Maclaurin series for $e^x$, manipulate it algebraically to form the function, differentiate it to generate the factor $n$, and plug in a specific value for $x$.

We know the Maclaurin series for $e^x$:
\[
e^x = \sum_{n=0}^{\infty} \frac{x^n}{n!} = 1 + x + \frac{x^2}{2!} + \frac{x^3}{3!} + \dots
\]
Subtract 1 and divide by $x$:
\begin{align*}
\frac{e^x-1}{x} &= \frac{1}{x} \left( \sum_{n=0}^{\infty} \frac{x^n}{n!} - 1 \right) \\
&= \frac{1}{x} \left( \sum_{n=1}^{\infty} \frac{x^n}{n!} \right) \\
&= \sum_{n=1}^{\infty} \frac{x^{n-1}}{n!}
\end{align*}
To match the denominator $(n+1)!$ in the target expression, let's shift the index. Let $k = n-1 \implies n = k+1$. When $n=1$, $k=0$:
\[
\frac{e^x-1}{x} = \sum_{k=0}^{\infty} \frac{x^k}{(k+1)!}
\]
We want to generate an $n$ (or $k$) in the numerator. We can do this by taking the derivative of both sides with respect to $x$.
On the right side (differentiating term-by-term):
\[
\frac{d}{dx} \left[ \sum_{k=0}^{\infty} \frac{x^k}{(k+1)!} \right] = \sum_{k=1}^{\infty} \frac{k x^{k-1}}{(k+1)!}
\]
On the left side (using the Quotient Rule):
\[
\frac{d}{dx} \left( \frac{e^x-1}{x} \right) = \frac{e^x(x) - (e^x-1)(1)}{x^2} = \frac{xe^x - e^x + 1}{x^2}
\]
Equating both sides:
\[
\frac{xe^x - e^x + 1}{x^2} = \sum_{k=1}^{\infty} \frac{k x^{k-1}}{(k+1)!}
\]
To evaluate the series $\sum_{k=1}^{\infty} \frac{k}{(k+1)!}$, we simply plug in $x=1$:
\[
\sum_{k=1}^{\infty} \frac{k (1)^{k-1}}{(k+1)!} = \frac{(1)e^1 - e^1 + 1}{1^2} = \frac{e - e + 1}{1} = 1
\]
Thus, $\sum_{n=1}^{\infty}\frac{n}{(n+1)!} = 1$.
\end{solution}

\begin{exercise}
Evaluate the following series.
\begin{enumerate}
    \item $\displaystyle \sum_{n=1}^{\infty} (-1)^{n-1}\frac{n}{2^n}$
    \item $\displaystyle \sum_{n=2}^{\infty} (-1)^{n}\frac{1}{2^n(n^2-1)}$
\end{enumerate}
\end{exercise}

\begin{solution}
\textbf{(1)} \textbf{Idea.} Construct a function $f(x)$ whose series expansion matches the given sum when $x = 1/2$. We will use integration to remove the $n$ in the numerator, identify the resulting geometric series, and then differentiate back to find $f(x)$.

Let $f(x) = \sum_{n=1}^{\infty} (-1)^{n-1} n x^n$. Our goal is to find $f(1/2)$.
To eliminate the $n$, we divide by $x$ to get $x^{n-1}$, which is the exact derivative of $x^n$.
Let $g(x) = \frac{f(x)}{x} = \sum_{n=1}^{\infty} (-1)^{n-1} n x^{n-1}$.
Now, integrate $g(x)$ term-by-term:
\begin{align*}
\int g(x) \,dx &= \int \left( \sum_{n=1}^{\infty} (-1)^{n-1} n x^{n-1} \right) dx \\
&= \sum_{n=1}^{\infty} (-1)^{n-1} \int n x^{n-1} \,dx \\
&= \sum_{n=1}^{\infty} (-1)^{n-1} x^n + C \\
&= x - x^2 + x^3 - x^4 + \dots + C
\end{align*}
This is a geometric series with first term $a=x$ and common ratio $r=-x$.
\[
\int g(x) \,dx = \frac{x}{1 - (-x)} + C = \frac{x}{1+x} + C
\]
To find $g(x)$, we differentiate both sides:
\[
g(x) = \frac{d}{dx} \left( \frac{x}{1+x} + C \right) = \frac{1(1+x) - x(1)}{(1+x)^2} = \frac{1}{(1+x)^2}
\]
Since $g(x) = f(x)/x$, we have:
\[
f(x) = x \cdot g(x) = \frac{x}{(1+x)^2}
\]
Finally, substitute $x = 1/2$ into $f(x)$ to get the sum of the series:
\[
\sum_{n=1}^{\infty} (-1)^{n-1}\frac{n}{2^n} = f\left(\frac{1}{2}\right) = \frac{1/2}{(1 + 1/2)^2} = \frac{1/2}{9/4} = \frac{2}{9}.
\]

\textbf{(2)} \textbf{Idea.} Construct $f(x) = \sum_{n=2}^{\infty} (-1)^n \frac{1}{n^2-1} x^n$, so the answer is $f(1/2)$. We will multiply by $x$, differentiate to remove $(n+1)$ from the denominator, and then integrate to reconstruct the function.

Note that $n^2 - 1 = (n-1)(n+1)$.
Let $f(x) = \sum_{n=2}^{\infty} (-1)^n \frac{1}{(n-1)(n+1)} x^n$.
Multiply by $x$ to match the power $n+1$ with the denominator:
\[
g(x) = xf(x) = \sum_{n=2}^{\infty} (-1)^n \frac{1}{(n-1)(n+1)} x^{n+1}
\]
Differentiate $g(x)$:
\begin{align*}
g'(x) &= \sum_{n=2}^{\infty} (-1)^n \frac{n+1}{(n-1)(n+1)} x^n = \sum_{n=2}^{\infty} (-1)^n \frac{1}{n-1} x^n \\
&= \sum_{n=2}^{\infty} (-1)^n \frac{1}{n-1} x \cdot x^{n-1} \\
&= x \sum_{n=2}^{\infty} (-1)^n \frac{x^{n-1}}{n-1}
\end{align*}
Let $k = n-1$. When $n=2, k=1$:
\[
g'(x) = x \sum_{k=1}^{\infty} (-1)^{k+1} \frac{x^k}{k}
\]
Recall the Maclaurin series for $\ln(1+x) = \sum_{k=1}^{\infty} (-1)^{k-1} \frac{x^k}{k}$. Since $(-1)^{k+1} = (-1)^{k-1}$, we have:
\[
g'(x) = x \ln(1+x)
\]
Now, integrate $g'(x)$ to find $g(x)$. Use integration by parts: let $u = \ln(1+x), dv = x dx \implies du = \frac{1}{1+x}dx, v = \frac{x^2}{2}$.
\begin{align*}
g(x) = \int x \ln(1+x) \,dx &= \frac{x^2}{2}\ln(1+x) - \frac{1}{2} \int \frac{x^2}{1+x} \,dx \\
&= \frac{x^2}{2}\ln(1+x) - \frac{1}{2} \int \frac{x^2 - 1 + 1}{1+x} \,dx \\
&= \frac{x^2}{2}\ln(1+x) - \frac{1}{2} \int \left( x - 1 + \frac{1}{1+x} \right) \,dx \\
&= \frac{x^2}{2}\ln(1+x) - \frac{1}{2} \left( \frac{x^2}{2} - x + \ln(1+x) \right) + C \\
&= \frac{1}{2}(x^2 - 1)\ln(1+x) - \frac{1}{4}x^2 + \frac{x}{2} + C
\end{align*}
Since $g(x) = xf(x)$ has its lowest degree term starting at $x^3$, $g(0) = 0$.
Evaluating at $x=0$: $g(0) = \frac{1}{2}(-1)\ln(1) - 0 + 0 + C = C \implies C = 0$.
So $g(x) = \frac{1}{2}(x^2 - 1)\ln(1+x) - \frac{1}{4}x^2 + \frac{x}{2}$.
Since $f(x) = \frac{g(x)}{x}$:
\[
f(x) = \frac{x^2 - 1}{2x}\ln(1+x) - \frac{1}{4}x + \frac{1}{2}
\]
Finally, evaluate at $x = 1/2$:
\begin{align*}
f(1/2) &= \frac{(1/4) - 1}{1} \ln(3/2) - \frac{1}{8} + \frac{1}{2} \\
&= -\frac{3}{4} \ln(3/2) + \frac{3}{8} = \frac{3}{8} - \frac{3}{4} \ln\left(\frac{3}{2}\right)
\end{align*}
\end{solution}

\begin{exercise}
Let $a_n=c+(n-1)d, n=1,2,\dots$, and $b>1$. Find the value of $\displaystyle \sum_{n=1}^{\infty}\frac{a_n}{b^n}.$
\end{exercise}

\begin{solution}
\textbf{Idea.} Split the sum into two parts. The first is a standard geometric series. The second requires differentiating a geometric series to generate the $(n-1)$ term.

\[
\sum_{n=1}^{\infty} \frac{c+(n-1)d}{b^n} = c \sum_{n=1}^{\infty} \frac{1}{b^n} + d \sum_{n=1}^{\infty} \frac{n-1}{b^n} = I + II
\]
For $I$, it is a geometric series with first term $a = 1/b$ and common ratio $r = 1/b$. Since $b>1$, $|r| < 1$.
\[
I = c \cdot \frac{1/b}{1 - 1/b} = c \cdot \frac{1/b}{(b-1)/b} = \frac{c}{b-1}
\]
For $II$, the $n=1$ term is 0, so we can start the sum from $n=2$:
\[
II = d \sum_{n=2}^{\infty} \frac{n-1}{b^n}
\]
Let's define a function $f(x) = \sum_{n=2}^{\infty} \frac{n-1}{b^n} x^{n-2}$, so that $II = d \cdot f(1)$.
We integrate $f(x)$ to remove the $(n-1)$ multiplier:
\begin{align*}
\int_0^x f(t) \,dt &= \sum_{n=2}^{\infty} \frac{n-1}{b^n} \int_0^x t^{n-2} \,dt = \sum_{n=2}^{\infty} \frac{n-1}{b^n} \left[ \frac{t^{n-1}}{n-1} \right]_0^x \\
&= \sum_{n=2}^{\infty} \frac{x^{n-1}}{b^n} = \frac{1}{b} \sum_{n=2}^{\infty} \left(\frac{x}{b}\right)^{n-1}
\end{align*}
This is a geometric series with first term $a = x/b^2$ (when $n=2$) and common ratio $r = x/b$.
\[
\int_0^x f(t) \,dt = \frac{1}{b} \cdot \frac{x/b}{1 - x/b} = \frac{1}{b} \cdot \frac{x/b}{(b-x)/b} = \frac{x}{b(b-x)}
\]
To find $f(x)$, we differentiate:
\[
f(x) = \frac{d}{dx} \left[ \frac{x}{b(b-x)} \right] = \frac{1}{b} \frac{1(b-x) - x(-1)}{(b-x)^2} = \frac{1}{b} \frac{b}{(b-x)^2} = \frac{1}{(b-x)^2}
\]
Thus, $II = d \cdot f(1) = \frac{d}{(b-1)^2}$.
Combining both parts:
\[
\sum_{n=1}^{\infty}\frac{a_n}{b^n} = I + II = \frac{c}{b-1} + \frac{d}{(b-1)^2} = \frac{c(b-1)+d}{(b-1)^2}.
\]
\end{solution}
